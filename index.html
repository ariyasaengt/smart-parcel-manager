โค้ด Google Apps Script เชื่อม Google Sheets

นำโค้ดด้านล่างไปวางใน Extensions > Apps Script ของ Google Sheets แล้ว Deploy เป็น Web App (เลือกสิทธิ์เป็น Anyone):

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (data.action === "save_batch") {
    data.items.forEach(function(item) {
      sheet.appendRow([item.id, item.dormName, item.target, item.courier, item.cod, item.date, item.status]);
    });
    return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
  }
}
